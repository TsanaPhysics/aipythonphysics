PlayCanvas Engine

ยกระดับห้องปฏิบัติการเสมือนจริงด้วยกราฟิกคุณภาพสูง ระบบอนุภาค และการปฏิสัมพันธ์ที่ลื่นไหล

1. Kinetic Molecular Theory (Gas particles)

การจำลองอนุภาคแก๊สในภาชนะปิด โดยใช้ระบบอนุภาค (Particle System) ของ PlayCanvas เพื่อแสดงการพุ่งชนและอุณหภูมิที่สัมพันธ์กับความเร็ว

[ 3D Visualization Placeholder: กล่องโปร่งใสที่มีลูกบอลขนาดเล็กพุ่งชนกันและเปลี่ยนสีตามอุณหภูมิ ]

การควบคุมระบบอนุภาคผ่าน Script

// ใน PlayCanvas Editor: สร้าง Particle System
// ใน Script: ปรับแต่งค่าตามอุณหภูมิ (T)
var GasParticles = pc.createScript('gasParticles');

GasParticles.prototype.setTemperature = function(temp) {
    this.entity.particlesystem.velocityGraph = new pc.Curve([0, temp * 0.5, 1, temp]);
    this.entity.particlesystem.colorGraph = new pc.CurveSet([
        [0, 0.5], [1, 1], // Red channel (ร้อนขึ้นจะแดงขึ้น)
        [0, 0.5], [1, 0.2] // Blue channel
    ]);
};

หัวข้อการเรียนรู้

นักศึกษาสามารถทดลองปรับเพิ่มอุณหภูมิในหน้าจอ แล้วสังเกตการเปลี่ยนแปลงของความดัน (จำลองผ่านจำนวนครั้งที่มีการชนผนัง) ซึ่งสอดคล้องกับกฎของแก๊สอุดมคติ (PV = nRT)

2. Virtual Microscope Viewport

การสร้างหน้าจอแสดงผลหน้ากล้องจุลทรรศน์ที่มีระบบซูม (Magnification) และการปรับความคมชัด (Fine Adjust) ผ่าน UI ของ PlayCanvas

[ 3D Visualization Placeholder: หน้าจอ interface ที่มีวงกลมส่องดูเซลล์แบคทีเรียที่ขยายใหญ่ขึ้น ]

สคริปต์การซูมด้วย FOV และตำแหน่งกล้อง

Microscope.prototype.zoom = function(level) {
    var targetFieldOfView = 45 / level;
    
    // Tween การเปลี่ยนแปลง FOV ให้ลื่นไหล
    this.entity.camera.calculateFov = function() {
        return pc.math.lerp(this.fov, targetFieldOfView, 0.1);
    };
    
    // แสดงผลข้อมูลกำลังขยายบน HTML UI
    document.getElementById('zoom-info').innerText = '100x (Oil Immersion)';
};
